JSONArray] ; ["id"; If(ValueCount(iDfieldType)>0; "[\"" & Substitute ( iDfieldType; "¶"; "\",\"") & "\"]" ; "[]" ) ; JSONArray] ) ) ]

#Custom Options for this widget
